rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Users can read their own profile.
    match /users/{userId} {
      allow read, write: if request.auth != null
                         && request.auth.uid == userId;
    }

    // Signed-in users can view seat occupancy. Only a new seat document
    // can be created; existing seat documents cannot be overwritten.
    match /bookedSeats/{seatDocId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
                    && request.resource.data.userId == request.auth.uid
                    && request.resource.data.status == "Booked";
      allow update, delete: if false;
    }

    // Each user can create and read their own booking documents.
    match /bookings/{bookingId} {
      allow create: if request.auth != null
                    && request.resource.data.userId == request.auth.uid;
      allow read: if request.auth != null
                  && resource.data.userId == request.auth.uid;
      allow update, delete: if false;
    }
  }
}
